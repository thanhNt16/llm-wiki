```python
def _page_graph_findings(wiki, add):
    edges = deps.graph(wiki).get("edges", {})
    page_arts = sorted(a for a in edges if a.startswith("wiki/")
                       and a.count("/") == 2 and a.endswith(".md"))
    # include pages on disk that have no edges yet (still orphan candidates)
    for kind in synthesize.PAGE_KINDS:
        d = wiki.p("wiki", kind)
        if os.path.isdir(d):
            for n in os.listdir(d):
                rel = "wiki/%s/%s" % (kind, n)
                if n.endswith(".md") and n != "index.md" and rel not in page_arts:
                    page_arts.append(rel)
    page_set = set(page_arts)
    out_e = {a: [d for d in edges.get(a, []) if d in page_set] for a in page_arts}
    in_e = {a: [] for a in page_arts}
    for a, ds in out_e.items():
        for d in ds:
            if d in in_e: in_e[d].append(a)
    for a in page_arts:
        if not in_e[a]:
            add("warn", "orphan_page", "%s has no inbound page links" % a,
                "link it from a related page or a question page")
        if not out_e[a]:
            add("warn", "dead_end_page", "%s links to no other page" % a,
                "add a See also section with [[wiki:kind/slug]] links")
    # walled gardens: undirected components, all edges internal, size>=3
    #   (union-find over page_set)
    # near-duplicate subjects: read each page frontmatter 'subject';
    #   Jaccard(norm_tokens(s1), norm_tokens(s2)) >= 0.7 and no edge either way
    # stale_page_ttl: frontmatter 'stale_after' (YYYY-MM-DD) < today
```

Implement union-find (~20 lines), Jaccard via `norm_tokens`, `stale_after` compare via `datetime.date.today().isoformat()`.
